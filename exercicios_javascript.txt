let idade = Number(prompt("Digite a idade do nadador:"));

if (idade >= 5 && idade <= 7) {
    alert("Categoria: Infantil A");
} else if (idade >= 8 && idade <= 10) {
    alert("Categoria: Infantil B");
} else if (idade >= 11 && idade <= 13) {
    alert("Categoria: Juvenil A");
} else if (idade >= 14 && idade <= 17) {
    alert("Categoria: Juvenil B");
} else if (idade >= 18) {
    alert("Categoria: Adulto");
} else {
    alert("Idade fora das categorias!");
}



let angulo = Number(prompt("Digite a medida do ângulo:"));

if (angulo === 0 || angulo === 180) {
    alert("Ângulo raso");
} else if (angulo > 0 && angulo < 90) {
    alert("Ângulo agudo");
} else if (angulo === 90) {
    alert("Ângulo reto");
} else if (angulo > 90 && angulo < 180) {
    alert("Ângulo obtuso");
} else if (angulo > 180 && angulo < 360) {
    alert("Ângulo côncavo");
} else if (angulo === 360) {
    alert("Ângulo completo");
} else {
    alert("Ângulo inválido!");
}




let codigo = Number(prompt("Digite o código do produto:"));
let quantidade = Number(prompt("Digite a quantidade:"));

let produto;
let preco;

switch (codigo) {
    case 200:
        produto = "Camiseta";
        preco = 25;
        break;
    case 201:
        produto = "Calça";
        preco = 35;
        break;
    case 202:
        produto = "Jaqueta";
        preco = 45;
        break;
    case 203:
        produto = "Saia";
        preco = 55;
        break;
    default:
        alert("Código inválido!");
}

if (preco !== undefined && quantidade > 0) {
    let total = preco * quantidade;

    alert(
        "Produto: " + produto +
        "\nQuantidade: " + quantidade +
        "\nValor total: R$ " + total.toFixed(2)
    );
}
